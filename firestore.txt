rules_version = '2';
service cloud.firestore {
  match /databases/{db}/documents {
    function member(f) { return request.auth != null && exists(/databases/$(db)/documents/families/$(f)/members/$(request.auth.uid)); }
    match /users/{u} { allow read, write: if request.auth != null && request.auth.uid == u; }
    match /families/{f} {
      allow get: if request.auth != null;
      allow list: if false;
      allow create: if request.auth != null && request.resource.data.owner == request.auth.uid && request.resource.data.keys().hasOnly(['owner','created']);
      allow update, delete: if false;
      match /members/{u} {
        allow read: if member(f);
        allow create: if request.auth != null && request.auth.uid == u && exists(/databases/$(db)/documents/families/$(f));
        allow delete: if request.auth != null && request.auth.uid == u;
      }
      match /items/{i} { allow read, write: if member(f) && (request.method == 'delete' || request.resource.size() < 200000); }
      match /cfg/{c}   { allow read, write: if member(f); }
    }
  }
}
